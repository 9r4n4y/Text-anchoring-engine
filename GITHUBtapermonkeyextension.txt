  function gm(method, url, body, tok, raw) {

        data: body ? JSON.stringify(body) : undefined,
        onload: (r) => {
          const ok = r.status >= 200 && r.status < 300;
          let data = null;
          if (raw && ok) data = r.response;
          else {
            let t = r.responseText;

            try { data = JSON.parse(t); } catch (_) { /* not json */ }
          }
          resolve({ status: r.status, data, headers: r.responseHeaders || '' });
        },
        onerror: () => reject(new Error('Network error. Check your connection and try again.')),
        ontimeout: () => reject(new Error('GitHub took too long to respond.')),

  async function api(method, path, body, opts = {}) {
    const tok = token();
    if (!tok) { const e = new Error('No token saved.'); e.status = 401; throw e; }
    const retries = opts.retries == null ? 4 : opts.retries;
    for (let attempt = 0; ; attempt++) {
      const r = await gm(method, API + path, body, tok, opts.raw);
      if (r.status >= 200 && r.status < 300) return r.data;
      const msg = (r.data && r.data.message) || `HTTP ${r.status}`;
      const limited = r.status === 429 || (r.status === 403 && /rate limit|abuse|secondary/i.test(msg));
      if (limited && attempt < retries) {
        const ra = /retry-after:\s*(\d+)/i.exec(r.headers);
        const secs = ra ? +ra[1] : Math.pow(2, attempt) * 4;
        if (onWait) onWait(secs);
        await sleep(secs * 1000);
        continue;
      }
      const err = new Error(msg);
      err.status = r.status;
      throw err;
    }
  }

      if (s) { c.st.set(p, s); (s === 'new' ? c.adds : s === 'mod' ? c.mods : s === 'copy' ? c.copies : c.moves).push(p); mark(p); }

    for (const p of B.blobs.keys()) if (!W.has(p)) { c.dels.push(p); mark(p); }
    c.total = c.adds.length + c.mods.length + c.moves.length + c.copies.length + c.dels.length;
    CHG = c;
    return c;
  }

  /* every user action goes through mutate() so it can be undone */
  const setW = (p, e) => { CUR.push({ p, prev: W.get(p), next: e }); if (e) W.set(p, e); else W.delete(p); };
  function mutate(label, fn) {
    CUR = [];
    try { fn(); } finally { const ch = CUR; CUR = null; if (ch.length) { undoSt.push({ label, ch }); redoSt.length = 0; } }
    pruneSel();
    bump();
  }
  function undo() {
    const o = undoSt.pop();
    if (!o) return toast('Nothing to undo');
    for (let i = o.ch.length - 1; i >= 0; i--) { const c = o.ch[i]; if (c.prev) W.set(c.p, c.prev); else W.delete(c.p); }
    redoSt.push(o); pruneSel(); bump(); toast(`Undid “${o.label}”`);
  }
  function redo() {
    const o = redoSt.pop();
    if (!o) return toast('Nothing to redo');
    for (const c of o.ch) { if (c.next) W.set(c.p, c.next); else W.delete(c.p); }
    undoSt.push(o); pruneSel(); bump(); toast(`Redid “${o.label}”`);
  }
  function pruneSel() {
    const keep = new Set();
    for (const p of S.sel) if (exists(p)) keep.add(p);
    S.sel = keep;
    if (S.anchor && !keep.has(S.anchor)) S.anchor = null;
    if (!isDir(S.cwd)) { S.cwd = ''; }
  }

  async function pool(items, n, worker) {
    let i = 0;
    let failed = false;
    const run = async () => {
      while (i < items.length && !failed) {
        const k = i++;
        try { await worker(items[k], k); } catch (e) { failed = true; throw e; }
      }
    };
    await Promise.all(Array.from({ length: Math.min(n, items.length) }, run));
  }

  async function transfer(paths, destDir, copy) {

      let dest = join(destDir, name);
      let replace = false;
      if (!copy && dirname(src) === destDir) continue;
      if (isDir(src) && (destDir === src || destDir.startsWith(`${src}/`))) { toast('A folder can’t be moved into itself.', { type: 'err' }); continue; }

        const v = await askConflict(name, state, paths.length > 1);

        if (v === 'skip') continue;
        if (v === 'both') dest = join(destDir, uniqueName(destDir, name));
        else replace = true;
      }
      plan.push({ src, dest, replace });
    }
    if (stopped && !plan.length) return;
    if (!plan.length) return;
    const label = `${copy ? 'Copy' : 'Move'} ${plan.length === 1 ? `“${basename(plan[0].src)}”` : plural(plan.length, 'item')}`;

          const e = W.get(k);

  }

      const file = await new Promise((res, rej) => entry.file(res, rej));

  /* ───────────────────────── download (file or ZIP) ───────────────────────── */
  const CRC = (() => { const t = new Uint32Array(256); for (let n = 0; n < 256; n++) { let c = n; for (let k = 0; k < 8; k++) c = c & 1 ? 0xEDB88320 ^ (c >>> 1) : c >>> 1; t[n] = c >>> 0; } return t; })();
  const crc32 = (u8) => { let c = -1; for (let i = 0; i < u8.length; i++) c = CRC[(c ^ u8[i]) & 255] ^ (c >>> 8); return (c ^ -1) >>> 0; };
  function makeZip(files) {
    const enc = new TextEncoder();
    const parts = []; const cen = []; let off = 0;
    const d = new Date();
    const dt = (d.getHours() << 11) | (d.getMinutes() << 5) | (d.getSeconds() >> 1);
    const dd = ((d.getFullYear() - 1980) << 9) | ((d.getMonth() + 1) << 5) | d.getDate();
    for (const f of files) {
      const nm = enc.encode(f.name); const crc = crc32(f.data); const sz = f.data.length;
      const lh = new DataView(new ArrayBuffer(30));
      lh.setUint32(0, 0x04034b50, true); lh.setUint16(4, 20, true); lh.setUint16(6, 0x0800, true); lh.setUint16(10, dt, true); lh.setUint16(12, dd, true);
      lh.setUint32(14, crc, true); lh.setUint32(18, sz, true); lh.setUint32(22, sz, true); lh.setUint16(26, nm.length, true);
      parts.push(lh.buffer, nm, f.data);
      const ch = new DataView(new ArrayBuffer(46));
      ch.setUint32(0, 0x02014b50, true); ch.setUint16(4, 20, true); ch.setUint16(6, 20, true); ch.setUint16(8, 0x0800, true); ch.setUint16(12, dt, true); ch.setUint16(14, dd, true);
      ch.setUint32(16, crc, true); ch.setUint32(20, sz, true); ch.setUint32(24, sz, true); ch.setUint16(28, nm.length, true); ch.setUint32(42, off, true);
      cen.push(ch.buffer, nm);
      off += 30 + nm.length + sz;
    }
    const cdSize = cen.reduce((a, b) => a + b.byteLength, 0);
    const end = new DataView(new ArrayBuffer(22));
    end.setUint32(0, 0x06054b50, true); end.setUint16(8, files.length, true); end.setUint16(10, files.length, true); end.setUint32(12, cdSize, true); end.setUint32(16, off, true);
    return new Blob([...parts, ...cen, end.buffer], { type: 'application/zip' });
  }
  function saveBlob(blob, name) {
    const url = URL.createObjectURL(blob);
    const a = h('a', { href: url, download: name });
    document.body.append(a); a.click(); a.remove();
    setTimeout(() => URL.revokeObjectURL(url), 30000);
  }
  async function downloadItems(paths) {
    paths = topLevel(paths);
    if (!paths.length) return;
    try {
      if (paths.length === 1 && W.has(paths[0])) {
        const p = paths[0];
        toast(`Downloading “${basename(p)}”…`, { id: 'dl', ms: 1500 });
        saveBlob(new Blob([await getBytes(W.get(p))]), basename(p));
        return;
      }
      await prepare(paths);
      const base = paths.length === 1 ? dirname(paths[0]) : S.cwd;
      const files = [];
      for (const p of paths) for (const k of under(p)) files.push(k);
      let done = 0;
      const out = new Array(files.length);
      toast(`Preparing ZIP… 0 / ${files.length}`, { id: 'dl', sticky: true });
      await pool(files, CONCURRENCY, async (k, i) => {
        out[i] = { name: base ? k.slice(base.length + 1) : k, data: await getBytes(W.get(k)) };
        toast(`Preparing ZIP… ${++done} / ${files.length}`, { id: 'dl', sticky: true });
      });
      const zname = paths.length === 1 ? basename(paths[0]) : `${S.repo.name}-${paths.length}-items`;
      saveBlob(makeZip(out), `${zname}.zip`);
      toast(`Saved ${zname}.zip (${plural(files.length, 'file')}).`, { id: 'dl', ms: 3500 });
    } catch (e) { toast(friendly(e), { id: 'dl', type: 'err' }); }
  }

  const ghUrl = (p, dir, ref) => { const { owner, name } = S.repo; return p ? `https://github.com/${owner}/${name}/${dir ? 'tree' : 'blob'}/${encRef(ref || S.branch)}/${encRef(p)}` : `https://github.com/${owner}/${name}/tree/${encRef(ref || S.branch)}`; };
  const rawUrl = (p) => `https://raw.githubusercontent.com/${S.repo.owner}/${S.repo.name}/${encRef(S.branch)}/${encRef(p)}`;
  function copyText(text, msg) {
    const done = () => toast(msg || 'Copied to clipboard.');
    try { if (typeof GM_setClipboard === 'function') { GM_setClipboard(text, 'text'); return done(); } } catch (_) { /* fall through */ }
    if (navigator.clipboard) navigator.clipboard.writeText(text).then(done, () => toast('Could not copy.', { type: 'err' }));
    else toast('Could not copy.', { type: 'err' });
  }
  function copyLink(kind, paths) {
    const list = paths.length ? paths : [S.cwd];
    const urls = list.map((p) => {
      const dir = isDir(p);
      if (kind === 'path') return p || '/';
      if (kind === 'raw') return dir ? ghUrl(p, true) : rawUrl(p);
      if (kind === 'perma') return ghUrl(p, dir, S.headSha);
      return ghUrl(p, dir);
    });
    const fresh = list.some((p) => chg().st.has(p) || chg().dirty.has(p));
    copyText(urls.join('\n'), fresh && kind !== 'path' ? 'Link copied. It works after you commit.' : (kind === 'path' ? 'Path copied.' : `${plural(urls.length, 'link')} copied.`));
  }

  function defaultMessage() {
    const c = chg();
    if (c.total === 1) {
      if (c.adds.length) return `Add ${basename(c.adds[0])}`;
      if (c.mods.length) return `Update ${basename(c.mods[0])}`;
      if (c.copies.length) return `Copy ${basename(c.copies[0])}`;
      if (c.moves.length) { const p = c.moves[0]; const f = W.get(p).from; return f && dirname(f) === dirname(p) ? `Rename ${basename(f)} to ${basename(p)}` : `Move ${basename(p)}`; }
      return `Delete ${basename(c.dels[0])}`;
    }
    return [c.adds.length && `Add ${c.adds.length}`, c.mods.length && `Update ${c.mods.length}`, c.moves.length && `Move ${c.moves.length}`, c.copies.length && `Copy ${c.copies.length}`, c.dels.length && `Delete ${c.dels.length}`].filter(Boolean).join(', ') + ' files';
  }

  async function runCommit({ msg, mode, newBranch }, prog) {
    const c = chg();
    const { owner, name } = S.repo;
    const base = `/repos/${owner}/${name}`;
    const branch = S.branch;
    onWait = (s) => prog.note(`GitHub asked us to slow down. Continuing in ${s}s…`);
    try {
      prog.step(0);
      const ref = await api('GET', `${base}/git/ref/heads/${encRef(branch)}`);
      const headSha = ref.object.sha;
      if (headSha !== S.headSha) {
        const r = await dialog({ title: 'The branch has new commits', msg: `“${branch}” changed on GitHub after you opened it. Committing now builds on the newest version, and files you moved or deleted may overwrite newer edits.`, buttons: [{ label: 'Cancel', value: null }, { label: 'Commit Anyway', value: 'go', primary: true }] });
        if (!r.value) return { cancelled: true };
      }
      const head = await api('GET', `${base}/git/commits/${headSha}`);
      const uploads = [...c.adds, ...c.mods];
      const shas = new Map();
      let done = 0;
      prog.step(1, 0, `0 / ${uploads.length}`);
      await pool(uploads, CONCURRENCY, async (p) => {
        const e = W.get(p);
        const body = e.text != null ? { content: e.text, encoding: 'utf-8' } : { content: await toB64(e.file), encoding: 'base64' };
        const b = await api('POST', `${base}/git/blobs`, body);
        shas.set(p, b.sha);
        prog.step(1, ++done / uploads.length, `${done} / ${uploads.length}`);
      });
      prog.step(2);
      const entries = [];
      for (const p of c.dels) { const b = B.blobs.get(p); entries.push({ path: p, mode: b.mode, type: b.type, sha: null }); }
      for (const p of [...c.moves, ...c.copies]) { const e = W.get(p); entries.push({ path: p, mode: e.mode, type: e.type, sha: e.sha }); }
      for (const p of uploads) { const e = W.get(p); entries.push({ path: p, mode: e.mode || '100644', type: 'blob', sha: shas.get(p) }); }
      let baseTree = head.tree.sha;
      for (let i = 0; i < entries.length; i += TREE_CHUNK) {
        const t = await api('POST', `${base}/git/trees`, { base_tree: baseTree, tree: entries.slice(i, i + TREE_CHUNK) });
        baseTree = t.sha;
        prog.step(2, Math.min(1, (i + TREE_CHUNK) / entries.length));
      }
      prog.step(3);
      const commit = await api('POST', `${base}/git/commits`, { message: msg, tree: baseTree, parents: [headSha] });
      prog.step(4);
      let finalBranch = branch;
      if (mode === 'new') { await api('POST', `${base}/git/refs`, { ref: `refs/heads/${newBranch}`, sha: commit.sha }); finalBranch = newBranch; }
      else await api('PATCH', `${base}/git/refs/heads/${encRef(branch)}`, { sha: commit.sha });
      prog.step(5);
      return { sha: commit.sha, branch: finalBranch, mode, count: c.total };
    } finally { onWait = null; }
  }

  async function openEditor(path, dir = S.cwd) {
    const isNew = path == null;
    let e = null; let original = '';
    if (isNew) { try { await ensureShallow(dir); } catch (err) { return toast(friendly(err), { type: 'err' }); } }
    else {
      e = W.get(path);
      if (!e || e.type === 'commit') return;
      if (e.size > MAX_EDIT_BYTES) return toast(`“${basename(path)}” is too large to edit here (2 MB limit). Download it instead.`, { type: 'warn' });
      toast('Opening…', { id: 'ed', ms: 800 });
      let u8;
      try { u8 = await getBytes(e); } catch (err) { return toast(friendly(err), { type: 'err' }); }
      if (u8.subarray(0, 8000).includes(0)) return toast('This looks like a binary file, so it can’t be edited as text.', { type: 'warn' });
      try { original = new TextDecoder('utf-8', { fatal: true }).decode(u8); } catch (_) { return toast('This file isn’t valid UTF-8 text, so it can’t be edited here.', { type: 'warn' }); }
    }
    const nameInp = isNew ? h('input', { class: 'inp', value: 'untitled.txt', spellcheck: 'false', style: 'max-width:300px' }) : null;
    const ta = h('textarea', { class: 'ed-ta', spellcheck: 'false', value: original });
    const gut = h('div', { class: 'ed-gut' });
    const stat = h('span', { class: 'sum' });
    const upd = () => {
      const n = ta.value.split('\n').length;
      if (gut.dataset.n !== String(n)) { gut.dataset.n = String(n); gut.textContent = Array.from({ length: n }, (_, i) => i + 1).join('\n'); }
      gut.scrollTop = ta.scrollTop;
      const before = ta.value.slice(0, ta.selectionStart).split('\n');
      stat.textContent = `Ln ${before.length}, Col ${before[before.length - 1].length + 1}  ·  ${ta.value.length.toLocaleString()} characters  ·  UTF-8`;
    };
    ['input', 'keyup', 'click', 'scroll'].forEach((ev) => ta.addEventListener(ev, upd));
    const close = async () => {
      if (ta.value !== original && !(isNew && !ta.value)) { const r = await dialog({ title: 'Discard your changes?', msg: 'Your edits to this file haven’t been saved.', buttons: [{ label: 'Keep Editing', value: null }, { label: 'Discard', value: 'go', danger: true }] }); if (!r.value) return; }
      closeSheet();
    };
    const save = async () => {
      const text = ta.value;
      if (isNew) {
        const n = cleanPath(nameInp.value);
        if (!n || !validPath(n)) return toast('Enter a valid file name.', { type: 'err' });
        const full = join(dir, n);
        let blocked = exists(full);
        for (let d = dirname(full); d && !blocked; d = dirname(d)) if (W.has(d)) blocked = true;
        if (blocked) return toast(`“${n}” already exists or clashes with a file.`, { type: 'err' });
        closeSheet();
        mutate(`New file “${basename(full)}”`, () => setW(full, newEntry(null, text)));
        if (dir !== S.cwd) await navigate(dir);
        const top = join(dir, n.split('/')[0]);
        S.sel = new Set([top]); S.anchor = top; ui.all();
        toast(`Added “${n}” to pending changes.`, { action: { label: 'Undo', fn: undo } });
      } else {
        if (text === original) return closeSheet();
        closeSheet();
        mutate(`Edit “${basename(path)}”`, () => setW(path, { ...W.get(path), text, file: null, dirty: true, size: new TextEncoder().encode(text).length }));
        toast(`Saved “${basename(path)}” to pending changes.`, { action: { label: 'Undo', fn: undo } });
      }
    };
    ta.addEventListener('keydown', (ev) => {
      if (ev.key === 'Tab') { ev.preventDefault(); const s = ta.selectionStart; ta.setRangeText('  ', s, ta.selectionEnd, 'end'); upd(); }
      else if ((ev.metaKey || ev.ctrlKey) && ev.key.toLowerCase() === 's') { ev.preventDefault(); save(); }
    });
    const wrapBtn = h('button', { class: 'xbtn', onClick: () => { ta.classList.toggle('wrap'); wrapBtn.textContent = ta.classList.contains('wrap') ? 'Wrap: On' : 'Wrap: Off'; } }, 'Wrap: Off');
    const head = h('div', { class: 'sh-h' }, isNew ? h('div', { class: 'big' }, icon('filePlus', 20)) : itemIcon({ name: basename(path), dir: false, e }, 32),
      h('div', { style: 'flex:1;min-width:0' }, isNew ? h('div', { class: 'fld' }, h('div', { class: 'sh-s' }, `New file in ${dir ? `“${dir}”` : S.repo.name}`), nameInp) : h('div', {}, h('div', { class: 'sh-t' }, basename(path)), h('div', { class: 'sh-s' }, path))));
    openSheet(h('div', { style: 'display:contents' }, head, h('div', { class: 'ed' }, gut, ta), h('div', { class: 'sh-f' }, stat, wrapBtn, h('button', { class: 'xbtn', onClick: close }, 'Cancel'), h('button', { class: 'xbtn pri', onClick: save }, isNew ? 'Add File' : 'Save'))), { wide: true, lock: true, onEsc: close });
    upd();
    setTimeout(() => (isNew ? nameInp : ta).focus(), 80);
    if (isNew) setTimeout(() => { nameInp.select(); }, 90);
  }

  }

  function onKey(e) {
    if (!S.open || S.min) return;

      else if (lk === 'i') { e.preventDefault(); toggleInspector(); }
      else if (lk === 's' || k === 'Enter') { e.preventDefault(); showCommitSheet(); }
      else if (k === '1') { e.preventDefault(); setView('icons'); }
      else if (k === '2') { e.preventDefault(); setView('list'); }
      else if (k === '[') { e.preventDefault(); goBack(); }
      else if (k === ']') { e.preventDefault(); goFwd(); }
      else if (k === '\\') { e.preventDefault(); toggleSidebar(); }
      else if (k === 'ArrowUp') { e.preventDefault(); goUp(); }
      else if (k === 'ArrowDown') { e.preventDefault(); const it = S.sel.size === 1 && (itemMap.get([...S.sel][0])); if (it) openItem(it); }
      else if (k === 'Backspace' || k === 'Delete') { e.preventDefault(); deleteItems([...S.sel]); }

    }

    else if (k === ' ') { e.preventDefault(); if (S.sel.size === 1) { const p = [...S.sel][0]; if (W.has(p)) openQuickLook(p); } }
    else if (k === 'Enter') { e.preventDefault(); if (S.sel.size === 1) beginRename([...S.sel][0]); }
